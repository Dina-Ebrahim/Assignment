db.createCollection("books", {

    $jsonSchema: {
      bsonType: "object",
      required: ["title"],
      properties: {
        title: {
          bsonType: "string",
          minLength: 1,
          description: "title must be a non-empty string"
        }
      }
    }
  }
})

//16

db.books.aggregate([
  {
    $match: {
      year: {
        $gt: 2000
      }
    }
  },
  {
    $sort: {
      year: -1
    }
  }
])


//17

db.books.aggregate([
  {
    $match: {
      year: {
        $gt: 2000
      }
    }
  },
  {
    $project: {
      _id: 0,
      title: 1,
      author: 1,
      year: 1
    }
  }
])


//18

db.books.aggregate([
  {
    $unwind: "$genres"
  }
])


//19

db.books.aggregate([
  {
    $lookup: {
      from: "logs",
      let: {
        bookId: { $toString: "$_id" }
      },
      pipeline: [
        {
          $match: {
            $expr: {
              $eq: ["$book_id", "$$bookId"]
            }
          }
        }
      ],
      as: "logs"
    }
  }
])
